cmake_minimum_required(VERSION 3.21)
file(READ "${CMAKE_CURRENT_LIST_DIR}/../../package.json" VIEZAN_PACKAGE_JSON)
string(JSON VIEZAN_VERSION GET "${VIEZAN_PACKAGE_JSON}" version)
project(ViezanNativeQt VERSION ${VIEZAN_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Svg Widgets Quick QuickControls2 Qml Network Concurrent Multimedia TextToSpeech WebSockets)
qt_standard_project_setup(REQUIRES 6.8)
qt_add_executable(ViezanNativeQt main.cpp WorkbenchBridge.cpp WorkbenchBridge.h WorkbenchDomain.cpp WorkbenchDomain.h LocalProvider.cpp LocalProvider.h CloudProtocol.cpp CloudProtocol.h CredentialStore.cpp CredentialStore.h NativeAudio.cpp NativeAudio.h Attachments.cpp Attachments.h LocalModelManager.cpp LocalModelManager.h NativeBackup.cpp NativeBackup.h ExtensionBridge.cpp ExtensionBridge.h DesktopIntegration.cpp DesktopIntegration.h WindowsSelection.cpp WindowsSelection.h ThermalMonitor.cpp ThermalMonitor.h Interpretation.cpp Interpretation.h LocalRuntime.cpp LocalRuntime.h ImageGeneration.cpp ImageGeneration.h Research.cpp Research.h ModelCatalog.cpp ModelCatalog.h LauncherTools.cpp LauncherTools.h SpeechSynthesis.cpp SpeechSynthesis.h AudioCapture.cpp AudioCapture.h EdgeSpeech.cpp EdgeSpeech.h CurrencyRates.cpp CurrencyRates.h NativeUpdates.cpp NativeUpdates.h LocalPerformance.cpp LocalPerformance.h)
if(WIN32)
    set(VIEZAN_ICON "${CMAKE_CURRENT_LIST_DIR}/../../build/icon.ico")
    configure_file(windows.rc.in "${CMAKE_CURRENT_BINARY_DIR}/windows.rc" @ONLY)
    target_sources(ViezanNativeQt PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/windows.rc")
endif()
set_target_properties(ViezanNativeQt PROPERTIES
    MACOSX_BUNDLE TRUE
    WIN32_EXECUTABLE TRUE
    MACOSX_BUNDLE_GUI_IDENTIFIER "com.viezan.native.qt.preview"
    MACOSX_BUNDLE_BUNDLE_NAME "Viezan")
file(GLOB NATIVE_ICONS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/qml/icons/*.svg")
set(NATIVE_RESOURCES qml/ImageTranslationView.qml qml/TranslationOptions.qml qml/EmptyState.qml qml/DesktopWorkspace.qml qml/Main.qml qml/RegionCapture.qml qml/HomeWorkspace.qml qml/DictionaryResult.qml qml/PrimaryNavigation.qml qml/SettingsWorkspace.qml qml/ProviderSettings.qml qml/LocalModelSettings.qml qml/DataSettings.qml qml/UsageSettings.qml qml/UpdateSettings.qml qml/BridgeSettings.qml qml/DesktopSettings.qml qml/DesktopPanels.qml qml/ThermalWorkspace.qml qml/InterpretationWorkspace.qml qml/ApplicationsWorkspace.qml qml/ComparisonPane.qml qml/NativeChart.qml qml/ChatTranscript.qml qml/native.json qml/brand.png ${NATIVE_ICONS})
qt_add_resources(ViezanNativeQt workbench PREFIX "/" FILES ${NATIVE_RESOURCES})
target_link_libraries(ViezanNativeQt PRIVATE Qt6::Svg Qt6::Widgets Qt6::Quick Qt6::QuickControls2 Qt6::Qml Qt6::Network Qt6::Concurrent Qt6::Multimedia Qt6::TextToSpeech Qt6::WebSockets)
install(TARGETS ViezanNativeQt RUNTIME DESTINATION bin BUNDLE DESTINATION .)

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 6.8 REQUIRED COMPONENTS Test)
    qt_add_executable(ViezanQtTests tests/WorkbenchBridgeTests.cpp WorkbenchBridge.cpp WorkbenchBridge.h WorkbenchDomain.cpp WorkbenchDomain.h LocalProvider.cpp LocalProvider.h CloudProtocol.cpp CloudProtocol.h CredentialStore.cpp CredentialStore.h NativeAudio.cpp NativeAudio.h Attachments.cpp Attachments.h LocalModelManager.cpp LocalModelManager.h NativeBackup.cpp NativeBackup.h ExtensionBridge.cpp ExtensionBridge.h DesktopIntegration.cpp DesktopIntegration.h WindowsSelection.cpp WindowsSelection.h ThermalMonitor.cpp ThermalMonitor.h Interpretation.cpp Interpretation.h LocalRuntime.cpp LocalRuntime.h ImageGeneration.cpp ImageGeneration.h Research.cpp Research.h ModelCatalog.cpp ModelCatalog.h LauncherTools.cpp LauncherTools.h SpeechSynthesis.cpp SpeechSynthesis.h AudioCapture.cpp AudioCapture.h EdgeSpeech.cpp EdgeSpeech.h CurrencyRates.cpp CurrencyRates.h NativeUpdates.cpp NativeUpdates.h LocalPerformance.cpp LocalPerformance.h)
    # Keep QtTest attached to the console so Windows CI captures failures.
    set_target_properties(ViezanQtTests PROPERTIES WIN32_EXECUTABLE FALSE MACOSX_BUNDLE FALSE)
    qt_add_resources(ViezanQtTests workbench_tests PREFIX "/" FILES ${NATIVE_RESOURCES})
    target_link_libraries(ViezanQtTests PRIVATE Qt6::Svg Qt6::Widgets Qt6::Quick Qt6::QuickControls2 Qt6::Qml Qt6::Network Qt6::Concurrent Qt6::Multimedia Qt6::TextToSpeech Qt6::WebSockets Qt6::Test)
    target_compile_definitions(ViezanQtTests PRIVATE VIEZAN_CONTRACT_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/../../contracts/native-workbench/recovery-fixture.json")
    add_test(NAME native-qt-bridge COMMAND ViezanQtTests)
    set_tests_properties(native-qt-bridge PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_CONTROLS_STYLE=Fusion")
endif()

foreach(native_target ViezanNativeQt ViezanQtTests)
    if(TARGET ${native_target})
        target_compile_definitions(${native_target} PRIVATE VIEZAN_VERSION="${PROJECT_VERSION}")
        if(WIN32)
            target_link_libraries(${native_target} PRIVATE Advapi32 User32 Ole32 OleAut32 Uiautomationcore Uuid Wintrust)
        elseif(UNIX AND NOT APPLE)
            find_package(Qt6 6.8 REQUIRED COMPONENTS DBus)
            target_link_libraries(${native_target} PRIVATE Qt6::DBus)
        elseif(APPLE)
            target_link_libraries(${native_target} PRIVATE "-framework Security" "-framework CoreFoundation")
        endif()
    endif()
endforeach()
